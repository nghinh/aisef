"""usage: python3 -B -I this.txt TREE STREAMS_LIST_FILE: usage.session_tokens of every listed stream under TREE."""
import json
import pathlib
import sys

tree, listing = pathlib.Path(sys.argv[1]), pathlib.Path(sys.argv[2])
sys.path.insert(0, str(tree))
from aisef2.app import usage  # noqa: E402

for rel in listing.read_text(encoding="utf-8").split("\n"):
    if rel:
        print(json.dumps({"path": rel, **usage.session_tokens(tree / rel)}, sort_keys=True))
