"""usage: python3 -B -I this.txt OUTDIR HEAD: writes OUTDIR/report.json from the final kc records."""
import json
import pathlib
import sys

OUT, HEAD = pathlib.Path(sys.argv[1]), sys.argv[2]


def kc(name):
    d = json.loads((OUT / f"kc-final/kc-{name}.json").read_text(encoding="utf-8"))
    n = len(d["results"])
    k = sum(r["class"] == "KILLED" for r in d["results"])
    return n, k, n - k
