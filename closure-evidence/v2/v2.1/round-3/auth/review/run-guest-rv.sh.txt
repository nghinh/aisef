# run-guest.sh TREE OUTDIR : linux_guest.sh's roles a b c d1 and the test modules, through $R3/guest.sh only

TREE=$1; D=$2; mkdir -p $D
ENTRY='mkdir -p /tmp/g; H=/repo/validation/v21/sandbox/linux_guest.py
