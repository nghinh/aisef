import ast, pathlib, sys, json
OUT = pathlib.Path(sys.argv[1])
src = (OUT / "exp/aisef2/runtime/process_range.py").read_text()
lines = src.splitlines(keepends=True)
tree = ast.parse(src)
targets = {"release", "_ladder", "_release", "__init__"}
found = []
for cls in [n for n in tree.body if isinstance(n, ast.ClassDef) and n.name in ("ProcessRange", "_Flight")]:
    for fn in [n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name in targets]:
        if cls.name == "ProcessRange" and fn.name == "__init__":
            sel = lambda n: n.lineno >= 318   # only the two new lines
        else:
            sel = lambda n: True
        for node in ast.walk(fn):
            if isinstance(node, (ast.Assign, ast.AugAssign, ast.AnnAssign, ast.Expr, ast.Return, ast.Raise)) and sel(node):
                if isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant):
                    continue  # docstring
                found.append((cls.name, fn.name, node))
manifest = []
for i, (c, f, node) in enumerate(found):
    s = lines[:]
    first = node.lineno - 1
    indent = lines[first][:len(lines[first]) - len(lines[first].lstrip())]
    # keep a statement's continuation lines out: replace the whole span by one `pass`
    s[first:node.end_lineno] = [indent + "pass\n"]
    name = f"s{i:02d}_{c}.{f}_L{node.lineno}"
    (OUT / "stmt" / f"{name}.py").write_text("".join(s))
    manifest.append({"name": name, "line": node.lineno, "text": lines[first].strip()[:90]})
json.dump(manifest, open(OUT / "stmt/manifest.json", "w"), indent=1)
print(len(manifest))
