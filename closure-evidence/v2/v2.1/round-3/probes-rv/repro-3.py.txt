# repro-3 (probes WP, item 3): file_artifact `python_imports` over a file that is valid Python but nests deeper than the
# compiler's stack.  Real entry points: FileArtifactProbe.observe and run_probe, over both readers (the object store of a
# real git repository, which is what `aisef run` uses, and a plain directory).  Also: the threshold on this interpreter,
# and the sibling facts of the same function (NUL byte, syntax error) for contrast.
# Run from the worktree root:  py.sh probes -I <this file>
import ast
import os
import pathlib
import subprocess
import sys
import tempfile

sys.path.insert(0, os.getcwd())

from aisef2.arch.enums import BehaviorVerdict, Enforcement, SubjectAbsence  # noqa: E402
from aisef2.probe import file_artifact as fa  # noqa: E402
from aisef2.probe.protocol import ExecutionEnv, RevisionRef, run_probe  # noqa: E402
from aisef2.product.spec import ProductProofSpec  # noqa: E402

ENV = ExecutionEnv(sys.executable, 30, Enforcement.FULL)
GIT_ENV = {k: os.environ[k] for k in ("PATH", "HOME", "TMPDIR") if k in os.environ}


def git(root, *args):
    return subprocess.run(["git", "-C", root, "-c", "user.name=t", "-c", "user.email=t@t", "-c", "commit.gpgsign=false",
                           *args], capture_output=True, text=True, check=True, env=GIT_ENV).stdout.strip()


def spec(matches=0):
    return ProductProofSpec.create(
        contract_id="BC-R3", probe_id=fa.PROBE_ID, probe_digest=fa.DIGEST,
        probe_input={"subject": {"kind": "file_artifact", "locator": "path:pkg"},
                     "stimulus": {"imports_outside": {"stdlib": True, "allowed": []}},
                     "observable": {"matches": matches, "within_s": 30},
                     "subject_absence": SubjectAbsence.REQUIRES_SUBJECT.value},
        candidate_expectation=BehaviorVerdict.SATISFIED, compiler_id="test", compiler_digest="c" * 64)


def tree(body: bytes, as_repo: bool):
    root = tempfile.mkdtemp(prefix="aisef2-r3-deep-")
    pathlib.Path(root, "pkg").mkdir()
    pathlib.Path(root, "pkg", "deep.py").write_bytes(body)
    if not as_repo:
        return RevisionRef("0" * 40, root)
    git(root, "init", "-q")
    git(root, "add", "-A")
    git(root, "commit", "-q", "-m", "x")
    return RevisionRef(git(root, "rev-parse", "HEAD"), root)


def show(label, body, as_repo):
    at = tree(body, as_repo)
    probe = fa.FileArtifactProbe()
    out = []
    for name, call in (("observe", lambda: probe.observe(spec(), at, ENV)),
                       ("run_probe", lambda: run_probe(probe, spec(), at, ENV))):
        try:
            r = call()
            if name == "observe":
                out.append("observe -> %s %s %s" % (r.kind.value, r.verdict.value if r.verdict else None, r.detail[-90:]))
            else:
                out.append("run_probe -> %r" % (r.result,))
        except BaseException as e:   # the point of the script: what escapes
            out.append("%s RAISED %s: %s" % (name, type(e).__name__, str(e)[:70]))
    print("%-34s %s" % (label, "repo " if as_repo else "plain"))
    for line in out:
        print("      " + line)


print("interpreter:", sys.version.split()[0], sys.platform, " file_artifact EXECUTION_DIGEST", fa._probe_digest()[:12])
# threshold of ast.parse on `x = 1+1+...+1` on this interpreter
last_ok, first_bad = 0, None
for n in (1000, 2000, 3000, 5000, 10000, 20000, 50000, 100000, 200000):
    try:
        ast.parse("x = 1" + "+1" * n)
        last_ok = n
    except RecursionError:
        first_bad = n
        break
    except BaseException as e:
        first_bad = "%s at %d" % (type(e).__name__, n)
        break
print("ast.parse('x = 1' + '+1'*N): largest N parsed in the sweep =", last_ok, "; first failing N =", first_bad)
try:
    ast.parse("x = 1" + "+1" * 100000)
except BaseException as e:
    print("ast.parse N=100000 raises:", type(e).__name__, str(e)[:80])

print()
deep = b"import os\nx = 1" + b"+1" * 100000 + b"\n"
show("valid, nested too deep", deep, True)
show("valid, nested too deep", deep, False)
show("control: syntax error", b"import os\nx = (\n", True)
show("control: NUL byte", b"import os\nx = 1\x00\n", True)
show("control: parses, one import", b"import os\nimport requests\n", True)
