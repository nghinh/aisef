import re, subprocess, sys
files = subprocess.run(["git", "ls-files", "-z"], capture_output=True, check=True).stdout.split(b"\0")
fault = re.compile(rb'step\\*"\s*:\s*\\*"fault\\*"')
sub = re.compile(rb'step\\*"\s*:\s*\\*"subprocess\\*"')
n_fault = 0; hits = []
for f in files:
    if not f: continue
    try:
        data = open(f, "rb").read()
    except OSError:
        continue
    fs = [m.start() for m in fault.finditer(data)]
    if not fs: continue
    n_fault += len(fs)
    for s in fs:
        window = data[s: s + 600]
        # the step that follows the fault in the same list: the next "step": value after this one
        nxt = re.search(rb'step\\*"\s*:\s*\\*"(\w+)\\*"', window[10:])
        if nxt and nxt.group(1) == b"subprocess":
            hits.append(f.decode())
print("fault step occurrences in tracked files:", n_fault)
print("fault immediately followed by a subprocess step:", sorted(set(hits)) or "none")
